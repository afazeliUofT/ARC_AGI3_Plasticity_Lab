def _find_player(g):
    cells = [(y, x) for y in range(BAR_ROW) for x in range(len(g[0]))
             if g[y][x] == BODY or g[y][x] == MARK]
    ys = [c[0] for c in cells]
    xs = [c[1] for c in cells]
    return min(ys), min(xs)


def _draw_player(g, top, left, facing):
    H = len(g)
    for dy in range(4):
        for dx in range(4):
            if top + dy < H:
                g[top + dy][left + dx] = BODY
    if facing == 'u':
        for dx in range(4):
            if top < H:
                g[top][left + dx] = MARK
    elif facing == 'd':
        for dx in range(4):
            if top + 3 < H:
                g[top + 3][left + dx] = MARK
    elif facing == 'l':
        for dy in range(4):
            if top + dy < H:
                g[top + dy][left] = MARK
    elif facing == 'r':
        for dy in range(4):
            if top + dy < H:
                g[top + dy][left + 3] = MARK


def _counted_actions(history, action):
    """Count actions that advance the timer: every action counts except a
    no-op (5) that immediately follows another no-op."""
    acts = [h["action"]["action"] for h in history[1:] if h.get("action")]
    acts.append(action.get("action"))
    count = 0
    prev = None
    for a in acts:
        if not (a == NOOP and prev == NOOP):
            count += 1
        prev = a
    return count


def predict(history, action):
    last = history[-1]
    g = copy.deepcopy(last["frame"][0])
    H, W = len(g), len(g[0])
    a = action.get("action")

    result = {
        "state": last["state"],
        "levels_completed": last["levels_completed"],
        "available_actions": list(last["available_actions"]),
    }

    if a in DIRS:
        top, left = _find_player(g)
        dy, dx = DIRS[a]
        new_facing = FACE[a]
        nt, nl = top + 4 * dy, left + 4 * dx
        old_cells = {(top + i, left + j) for i in range(4) for j in range(4)}
        ok = 0 <= nt and nt + 3 < H and 0 <= nl and nl + 3 < W
        if ok:
            for i in range(4):
                for j in range(4):

                    if (y, x) in old_cells or y >= BAR_ROW:
                        continue
                    if g[y][x] != BG:
                        ok = False
        for (y, x) in old_cells:
            if y < BAR_ROW:
                g[y][x] = BG
        if ok:
            top, left = nt, nl
        _draw_player(g, top, left, new_facing)

    # energy bar: one cell consumed every 3 counted actions (ticks at counted
    # indices 2, 5, 8, ...); a no-op directly after a no-op is not counted.
    count = _counted_actions(history, action)
    used = (count + 1) // 3
    for x in range(W):
        g[BAR_ROW][x] = BAR_BASE
    for k in range(min(used, W)):
        g[BAR_ROW][W - 1 - k] = BAR_USED

    result["frame"] = [g]
    return result
